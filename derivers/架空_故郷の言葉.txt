function roma2kata(s) {
  const r = /^([kgsztdnpbmylfvwyh]*[yw]*)([aiueo])([tniu]*)$/g; // 將音節分為韻頭、主要元音及韻尾
  const match = r.exec(s);
  if (match == null) {
    throw new Error('無法轉換為假名：' + s);
  }
  //const { 1: 韻頭, 2: 主要元音, 3: 韻尾 } = match;
  const [_ , 韻頭, 主要元音,  韻尾 ] = match;
  let 假名韻尾 = 韻尾表[韻尾];

  if (韻頭.indexOf("y") === -1&&韻頭.indexOf("w") === -1||韻頭 === "w"||韻頭 === "y") {
    return 假名表[韻頭 + 主要元音] + 假名韻尾;
  }
  const 填充元音 = 韻頭[韻頭.length-1] === 'w' ? 'u' : 'i'; 
  const 韻頭聲母 = 韻頭.slice(0,韻頭.length-1);
  return 假名表[韻頭聲母 + 填充元音] + 拗音表[韻頭[韻頭.length-1] + 主要元音] + 假名韻尾;
}

function kata2hira(s) {
  const diff = 'ぁ'.charCodeAt(0) - 'ァ'.charCodeAt(0);
  return [...s].map((c) => {
    switch (c) {
      case '𛅤': return '𛅐';
      case '𛅥': return '𛅑';
      case '𛅦': return '𛅒';
      default: return String.fromCharCode(c.charCodeAt(0) + diff);
    }
  }).join('');
}
