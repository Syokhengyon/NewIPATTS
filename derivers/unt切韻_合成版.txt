/* unt 切韻擬音 —— 合成版（synthesis build）
 *
 * 原方案：dersivers/unt切韻.txt（作者 unt，依 Wang, Tianheng & Xun Gong 2026,
 *   "Reconstructing pharyngealized vowels for Qieyun Division II in Middle Chinese",
 *   Diachronica. doi:10.1075/dia.25018.wan）
 *
 * 本檔只為「用本專案的 espeak 引擎念得出來、而且不用調任何選項」而改；
 * 音系內容一律不動，差別如下：
 *
 *   1. 聲調：改成數字調值 —— 平 44、上 45、去 41、入 4。
 *      原方案默認用組合附加符（U+0301 上聲 / U+0300 去聲），
 *      本引擎會把它們整段丟掉（實測 1990 次出現、100% 被丟棄），等於沒有聲調。
 *
 *   2. RTR 符號（U+0319，原方案加在非三等聲母下）不寫出 —— 同樣會被丟棄（1184 次）。
 *      但內部仍保留這個記號，因為「非三等聲母 → 小舌音」的轉換要靠它判斷
 *      （k̙→q、k̙ʰ→qʰ、ŋ̙→ɴ、x̙→χ、ɣ̙→ʁ）。
 *
 *   3. ATR 介音關閉（三 C 類不寫 ɰ，莊組三等不寫 ɹ）。重紐 A/B/C 靠原方案的
 *      類介音 j / ɹ / ∅ 區分，已經足夠。
 *
 *   4. 二等的咽化元音 ˤa ˤe ˤo → ħa ħe ħo。
 *      本引擎把 ˤ 映射成咽擦音 ?\，寫在元音前就成了「輔音 + 元音」；
 *      改成 ħ + 元音之後，元音的舌位完全不動（ˤ 只是咽化附加符），
 *      咽化的對立也還在。ħ 在本引擎的音素名是 H（見 src/ipa-espeak.mjs），
 *      實測不加時值、無喉塞停頓（二等 qħam 821ms，一等 qɑm 782ms）。
 *      （試過 ʕ：它映射成 ?\，會在元音前插 ~85ms 喉塞，故不用。）
 *
 *   5. ATR 介音打開、三C 介音寫成 ɯ（原方案的 ɰ 的元音形式）。
 *      為什麼不能關：非三等的 RTR 記號 U+0319 在本引擎會被丟掉，
 *      三等 C 類若又是零介音，宕攝就一三等相撞了 ——
 *      唐一「郎」lɑŋ 與 陽三C「良」會變成同一個字串。
 *
 *      為什麼不直接寫原方案的 ɰ 或 ɣ：
 *        ɰ → 本表映射成 r.，與三 B 的 ɹ 撞（同一個音素名）；
 *        ɣ / ɣ̞ → 映射成 g"（或 Q），實測跟在意音後面時會把元音推遲 ~75ms，
 *                 聽起來就是一個明顯的停頓（kɣɑŋ 元音 160ms 起，無介音才 80ms）。
 *        ɯ → 元音 80ms 起，無停頓；而且原方案自己就註明三C介音是
 *             「軟腭近音（ɨ~ɯ 對應的半元音）」，ɰ 正是 ɯ 的半元音寫法，
 *             寫成 ɯ 在音系上是同一個東西，只是本引擎發得乾淨。
 *      （同部位的介音會靠下面的「介音與元音同部位時省略」自動刪掉。）
 *
 *   6. 支韻固定 ie、微韻合口固定 uj（原方案默認值本來就是這兩個）。
 *
 *   7. 保留：見組非三等寫小舌音、音韻地位正則化、介音與元音同部位時省略。
 *
 * 原方案的可調項（RTR / 字母有降部時的RTR / ATR / 三C介音 / 見組非三等寫小舌音 /
 * 通江宕攝音節尾寫小舌音 / 後低元音寫a / 支韻 / 微韻合口 / 聲調記號 /
 * 音韻地位正則化 / 幫組拼ə時添加w介音）在本檔一律寫死，不再出現在界面上。
 *
 * @author unt（原方案）／本專案改寫為合成版
 */

/** @type { 音韻地位['屬於'] } */
const is = (...x) => 音韻地位.屬於(...x);
/** @type { 音韻地位['判斷'] } */
const when = (...x) => 音韻地位.判斷(...x);

/* 沒有設定項：直接返回空陣列（頁面會顯示「此方案沒有宣告設定項」） */
if (!音韻地位) return [];

/* 固定取值 */
const 寫小舌音 = true;      // 見組（含曉匣）非三等寫 q qʰ ɴ χ ʁ
const 寫ATR = true;         // 三等加 ATR 介音（非寫不可，見檔頭第 5 條）
const 三C介音 = 'ɯ';         // 三 C 類介音（原方案的 ɰ 寫成它的元音形式，見檔頭第 5 條）
const 支韻 = 'ie';
const 微韻合口 = 'uj';
const 調值 = { 平: '44', 上: '45', 去: '41', 入: '4' };

function 音韻地位正則化() {
  function 調整(表達式, 調整屬性) { if (is(表達式)) 音韻地位 = 音韻地位.調整(調整屬性); }
  [['微', '脂'], ['殷文', '真'], ['元', '仙'], ['嚴凡', '鹽']]
    .forEach(([C1, AB]) => 調整(`銳音 ${C1}韻`, `${AB}韻`));
  調整('銳音 幽韻', `尤韻 開合中立 ${is`端組` ? '四等' : ''}`);
  調整('明母 尤韻', '侯韻 一等 不分類');
}

function get聲母() {
  if (is`云母 開口 非 (深咸攝 入聲)`) return ''; // 云母開口爲零聲母
  let 聲母 = {
    幫: 'p', 滂: 'pʰ', 並: 'b', 明: 'm',
    端: 't', 透: 'tʰ', 定: 'd', 泥: 'n', 來: 'l',
    知: 'ʈ', 徹: 'ʈʰ', 澄: 'ɖ', 孃: 'ɳ',
    見: 'k', 溪: 'kʰ', 羣: 'ɡ', 疑: 'ŋ', 曉: 'x', 匣: 'ɣ', 云: 'w',
    影: 'ʔ',
    精: 'ts', 清: 'tsʰ', 從: 'dz', 心: 's', 邪: 'z',
    莊: 'tʂ', 初: 'tʂʰ', 崇: 'dʐ', 生: 'ʂ', 俟: 'ʐ',
    章: 'tɕ', 昌: 'tɕʰ', 常: 'dʑ', 書: 'ɕ', 船: 'ʑ', 日: 'ɲ', 以: 'j',
  }[音韻地位.母];
  /* 非三等的鈍音聲母先記上 RTR 記號（內部用，最後會移除） */
  if (is`(幫端精見影組 或 來母) 非 三等`) 聲母 = [...聲母].map(c => c === 'ʰ' ? c : c + '̙').join('');
  return 聲母;
}

function get介音() {
  let 類介音 = { A: 'j', B: 'ɹ' }[音韻地位.類] ?? '';
  let 呼介音 = is`合口 非 云母` ? 'w' : '';
  return 類介音 + 呼介音;
}

function get音節核尾() {
  let [韻列表, 核] = [
    ['脂　│　　　│幽　│　　│真臻　│侵　　', 'i'],
    ['之　│微　　│　　│蒸　│殷　　│　　　', 'ɨ'],
    ['尤侯│＿＿＿│＿＿│東＿│文＿＿│＿＿＿', 'u'],
    ['支　│齊祭　│蕭宵│青　│先仙　│鹽添　', 'e'],
    ['佳　│皆　　│　　│耕　│山　　│咸　　', 'ˤe'],
    ['魚　│灰咍廢│豪　│登　│元魂痕│覃嚴凡', 'ə'],
    ['虞模│　　　│　　│冬鍾│　　　│　　　', 'o'],
    ['＿＿│＿＿＿│＿＿│江＿│＿＿＿＿│＿＿', 'ˤo'],
    ['麻　│夬　　│肴　│庚清│刪　　│銜　　', is`二等` ? 'ˤa' : 'a'],
    ['歌　│泰　　│　　│陽唐│寒　　│談　　', 'ɑ'],
  ].find(e => e[0].includes(音韻地位.韻));
  let 尾 = ['', ...'jwŋnm'][韻列表.split('│').findIndex(e => e.includes(音韻地位.韻))];
  if (is`入聲`) 尾 = { ŋ: 'k', n: 't', m: 'p' }[尾];
  return { 核, 尾 };
}

function get聲調() {
  return 調值[音韻地位.聲];
}

function get音節() {
  const 音節 = {
    聲母: get聲母(),
    介音: get介音(),
    ...get音節核尾(),
    聲調: get聲調(),
  };
  /* 小舌音：靠內部 RTR 記號判斷（見上） */
  if (寫小舌音) 音節.聲母 = { k̙: 'q', k̙ʰ: 'qʰ', ŋ̙: 'ɴ', x̙: 'χ', ɣ̙: 'ʁ' }[音節.聲母] ?? 音節.聲母;
  /* RTR 記號一律不寫出 */
  音節.聲母 = 音節.聲母.replace(/̙/g, '');
  /* 三等的 ATR 介音（原方案的「ATR」選項，固定開啟）：三C → ɣ、莊組三等 → ɹ、其他銳音三等 → j */
  if (寫ATR) 音節.介音 = when([
    ['C類', 三C介音],
    ['莊組 三等', 'ɹ'],
    ['銳音 三等 非 以母', 'j'],
    ['', ''],
  ]) + 音節.介音;
  if (is`支韻`) 音節.核 = 支韻;
  if (is`微韻 非 開口`) 音節.核 = 微韻合口.slice(-2, -1);   // 'uj' → 'u'
  if (is`四等` && 音節.核 === 'a') 音節.介音 = 'j' + 音節.介音; // 爹小韻
  /* 介音與元音同部位時省略 */
  if (音節.核[0] === 'i') 音節.介音 = 音節.介音.replace('j', '');
  if (音節.核[0] === 'ɨ') 音節.介音 = 音節.介音.replace(三C介音, '');
  if (['u', 'o'].includes(音節.核[0])) 音節.介音 = 音節.介音.replace('w', '');
  音節.韻基 = 音節.核 + 音節.尾;
  音節.帶調韻母 = 音節.介音 + 音節.韻基 + 音節.聲調;
  return 音節;
}

if (音韻地位) 音韻地位正則化();
const 音節 = get音節();

/* 二等咽化：ˤ + 元音 → ħ + 元音（元音本身不動） */
return (音節.聲母 + 音節.帶調韻母).replace(/ˤ/g, 'ħ');
