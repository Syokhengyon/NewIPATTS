function 調整(表達式, 調整屬性) { if (is(表達式)) 音韻地位 = 音韻地位.調整(調整屬性); }

function 聲母規則() {
  if (is('幫母')) {
    if(['不'].includes(字頭)&&is('入聲'))return 'p';
    return is('東韻 三等 或 鍾微廢文元凡虞尤陽韻')?'f':'p';
  }
  if (is('滂母')) {
    return is('東韻 三等 或 鍾微廢文元凡虞尤陽韻')?'f':'pʰ';
  }
  if (is('並母')) {
    return is('東韻 三等 或 鍾微廢文元凡虞尤陽韻')?'f':is('平聲')?'pʰ':'p';
  }
  if (is('明母')) {
    return is('微廢文元凡虞陽韻')?'v':'m';
  }

  if (is('端母')) return 't';
  if (is('透母')) return 'tʰ';
  if (is('定母')) return is('平聲')?'tʰ':'t';
  if (is('泥母')) return 'n';
  if (is('來母 非 三等')) return 'l';

  if (is('知母')) return 'tʃ';
  if (is('徹母')) return 'tʃʰ';
  if (is('澄母')) return is('平聲')?'tʃʰ':'tʃ';
  if (is('孃母')) return 'n';
  if (is('來母')) return 'l';

  if (is('精母')) return 'ts';
  if (is('清母')) return 'tsʰ';
  if (is('從母')) return is('平聲')?'tsʰ':'ts';
  if (is('心母')) return 's';
  if (is('邪母')) return 's';

  if (is('莊母')) return 'tʃ';
  if (is('初母')) return 'tʃʰ';
  if (is('崇母')) return is('止攝')?'ʃ':is('平聲')?'tʃʰ':'tʃ';
  if (is('生母')) return 'ʃ';
  if (is('俟母')) return 'ʃ';

  if (is('章母')) return 'tʃ';
  if (is('昌母')) return 'tʃʰ';
  if (is('常母')) return is('陽聲韻 平聲')?'tʃʰ':'ʃ';
  if (is('書母')) return 'ʃ';
  if (is('船母')) return 'ʃ';
  if (is('日母')) return 'ʒ';

  if (is('見母')) return 'k';
  if (is('溪母')) return 'kʰ';
  if (is('羣母')) return is('平聲')?'kʰ':'k';
  if (is('疑母')) return 'ŋ';

  if (is('曉母')) return 'x';
  if (is('匣母')) return 'x';
  if (is('影母')) return '';
  if (is('云母')) return '';
  if (is('以母')) return '';

  throw new Error('無聲母規則');
}

function 韻母規則() {
  // 通攝
  if (is('東冬韻 一等')) return 'ʊŋ';
  if (is('東鍾韻 三等 孃母 入聲')) return 'iʊŋ';
  if (is('東鍾韻 三等')) return is('牙喉音 或 精組')?'iʊŋ':'ʊŋ';

  // 江攝
  if (is('江韻 牙喉音 非 影母')) return 'iaŋ';
  if (is('江韻 影母 入聲')) return 'uaŋ';
  if (is('江韻')) return is('知莊組')?'uaŋ':'aŋ';

  // 止攝
  if (is('支脂之韻 開口 章組')) return 'iɨ';
  if (is('支脂之微韻 開口')) return is('精莊組 或 日母')?'ɨ':'i';
  if (is('支脂之微韻 合口')) return is('莊組')?'uai':'uəi';
  if (is('支脂之韻')) return 'i';
  if (is('微韻')) return 'i';


  // 遇攝
  if (is('魚韻')) return is('莊組')?'ɔ':'yu';
  if (is('虞韻')) return is('莊組')?'ɔ':is('幫組')?'u':'yu';
  if (is('模韻')) return 'u';

  // 蟹攝
  if (is('齊韻')) return is('合口')?'uəi':'i';
  if (is('祭韻 莊組 開口')) return 'ɨ'
  if (is('祭韻 以母 合口')) return 'yi'
  if (is('祭韻')) return is('合口')?'uəi':'i';
  if (is('泰韻 疑母 合口')) return 'uai';
  if (is('泰韻')) return is('開口')?'ai':is('幫組')?'əi':'uəi';
  if (is('佳皆夬韻')&&蟹攝二等歸假攝字.includes(字頭)) return 'a';
  if (is('佳皆夬韻')) return 'ai';
  if (is('灰韻')) return is('合口')?'uəi':'əi';
  if (is('咍韻')) return 'ai';
  if (is('廢韻 開口')) return is('章組')?'iai':'i';
  if (is('廢韻')) return is('合口')?'uəi':'i';

  // 臻攝
  if (is('真韻')) return is('莊組')?'ən':'in';
  if (is('殷韻')) return 'in';
  if (is('臻韻')) return 'ən';
  if (is('文韻')) return is('幫組')?'ən':'in';
  if (is('元韻')) return is('幫莊組')?'an':'iɛn';
  if (is('魂韻 入聲 非 脣音')) return 'un';
  if (is('魂韻')) return 'ən';
  if (is('痕韻')) return 'ən';

  // 山攝
  if (is('寒韻')) return is('開口')?'an':is('牙喉音')?'uɔn':'ɔn';
  if (is('刪山韻')) return 'an';
  if (is('仙韻')) return is('莊組')?'an':'iɛn';
  if (is('先韻')) return 'iɛn';

  // 效攝
  if (is('蕭韻')) return 'iɛu';
  if (is('宵韻')) return 'iɛu';
  if (is('肴韻')) return 'au';
  if (is('豪韻')) return 'au';

  // 果攝
  if (is('歌韻 一等')&&果攝滯古.includes(字頭)) return 'a';
  if (is('歌韻 一等')) return is('牙喉音 合口')?'uɔ':'ɔ';
  if (is('歌韻 三等')) return is('幫組')?'a':'iɛ';

  // 假攝
  if (is('麻韻 二等')) return 'a';
  if (is('麻韻 三等')) return 'iɛ';
  if (is('麻韻 四等')) return 'iɛ';

  // 宕攝
  if (is('陽韻 幫組')) return 'aŋ';
  if (is('陽韻 合口')) return is('入聲')?'iaŋ':'uaŋ';
  if (is('陽韻')) return is('莊組')?'uaŋ':'iaŋ';
  if (is('唐韻')) return 'aŋ';

  // 梗攝
  if (is('庚耕韻 二等 匣母 開口 非 入聲')&&['行莖幸倖杏荇洐莕'].join('').includes(字頭)) return 'iŋ';
  if (is('庚耕韻 二等 影母 開口 非 入聲')) return 'iŋ';
  if (is('庚耕韻 端組')) return 'a';
  if (is('庚耕韻 二等')) return 'əŋ';
  if (is('庚韻 莊組')) return 'əŋ';
  if (is('庚韻 三等')) return 'iŋ';
  if (is('青韻')) return 'iŋ';
  if (is('清韻')) return 'iŋ';


  // 曾攝
  if (is('蒸韻')) return is('莊組')?'əŋ':'iŋ';
  if (is('登韻'))  return is('合口')?'uəŋ':'əŋ';

  // 流攝
  if (is('尤侯韻 明母')&&流攝明母歸遇攝字.includes(字頭)) return 'u';
  if (is('尤侯韻 明母')) return 'əu';
  if (is('尤侯韻 幫滂並母')&&流攝幫滂並母未歸遇攝字.includes(字頭)) return 'əu';
  if (is('尤侯韻 幫滂並母')) return 'u';
  if (is('尤韻')) return is('莊組')?'əu':'iəu';
  if (is('幽韻')) return is('幫滂並母')?'iɛu':'iəu';
  if (is('侯韻')) return 'əu';

  // 深攝
  if (is('侵韻')) return is('莊組')?'əm':'im';

  // 咸攝
  if (is('覃韻')) return 'am';
  if (is('談韻')) return 'am';
  if (is('凡韻')) return 'am';
  if (is('嚴鹽添韻')) return is('莊組')?'am':'iɛm';
  if (is('咸銜韻')) return 'am';

  throw new Error('無韻母規則');
}

function 聲調規則() {
  if (1) {
    if (is('全清 或 次清')) {
      if (is('平聲')) return '⁵⁵'; // 陰平
      if (is('上聲')) return '⁵³'; // 陰上
      if (is('去聲')) return '²⁴'; // 陰去
      if (is('入聲')) return '⁵'; // 陰入
    }
    else {
      if (is('平聲')) return '²¹²'; // 陽平
      if (is('全濁 上聲')) return '³³'; // 陽上
      if (is('上聲')) return '⁵³'; // 陽上
      if (is('去聲')) return '³³'; // 陽去
      if (is('次濁 入聲')) return '²'; // 陽入
      if (is('入聲')) return '²'; // 陽入
    }
  }

  throw new Error('無聲調規則');
}

let 聲母 = 聲母規則();
let 韻母 = 韻母規則();
let 聲調 = 聲調規則();

if (is('入聲')) {

  if (韻母.endsWith('m')) 韻母 = 韻母.slice(0, -1) + 'ʔ';
  else if (韻母.endsWith('n')) 韻母 = 韻母.slice(0, -1) + 'ʔ';
  else if (韻母.endsWith('ŋ')) 韻母 = 韻母.slice(0, -1) + 'ʔ';
}

if (is('合口')) {
  韻母='u'+韻母;
韻母=韻母.replace('uu','u');
韻母=韻母.replace('uy','y');
韻母=韻母.replace('uin','yin');
韻母=韻母.replace('uiŋ','yiŋ');
韻母=韻母.replace('uiʔ','yiʔ');
韻母=韻母.replace('ui','y');

if(is('歌韻 舌齒音 或 寒韻 入聲 舌齒音'))
{韻母=韻母.replace('uɔ','ɔ');}
}

韻母=韻母.replace('uʔ','ʊʔ');

if(is('牙喉音 開口 二等 非 梗江攝'))
{
  韻母='i'+韻母;
  if(is('蟹攝 溪影母'))
  {韻母=韻母.replace(/^i/,'');}
}

if(is('宕江攝 入聲'))
{
韻母=韻母.replace('a','ɔ');
韻母=韻母.replace('yɔ','iɔ');
}
if(is('寒覃談韻 牙喉音 入聲'))
{
韻母=韻母.replace('a','ɔ');
}
if(is('疑母'))
{
  if(韻母[0]==='i')
{
聲母=聲母.replace('ŋ','ȵ');
}
if(韻母[0]==='u'||韻母[0]==='y')
{
聲母=聲母.replace('ŋ','');
}

}

if(is('知章莊組 或 日母'))
{
if(韻母[0]==='i'||韻母[0]==='y')
{聲母=聲母.replace('ʃ','ɕ');
聲母=聲母.replace('ʒ','ʑ');
}
else
{
聲母=聲母.replace('ʃ','ʂ');
聲母=聲母.replace('ʒ','ʐ');
}
}
return 聲母 + 韻母 + 聲調;
