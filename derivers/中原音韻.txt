 * - 楊耐思. 中原音韻音系. 北京: 中國社會科學出版社, 1981.
 * - 寧繼福. 中原音韻表稿. 長春: 吉林文史出版社, 1985.
 * - 薛鳳生. 中原音韻音位系統. 魯國堯, 侍建國, 譯. 北京: 北京語言學院出版社, 1990.

 * ── 本版改動（精簡為固定設定，音值取 unt 2021）──
 *
 * 1. 設定項只留「包含部分例外音變」；其餘一律寫死：
 *      顯示      → 音值（unt, 2021）
 *      高元音開口呼 → ɨ
 *      皆來韻古入聲字韻基 → ɛj（實測與舒聲不衝突：舒聲 83 個音節、入聲 24 個，無一相同）
 *      皆來韻梗二開見入   → jaj（規則順序調到 ɛj 之前，否則會被寫成 jɛj）
 * 2. 聲調改用現代普通話的五度值：陰平 ⁵⁵、陽平 ³⁵、上聲 ²¹、去聲 ⁵¹
 *    （普通話標準的上聲是 ²¹⁴，但本引擎讀三位調不穩，使用者決定統一寫 ²¹）
 *    （原方案是調類數字 ¹²³⁴，不是調值）
 * 3. 古入聲字不再加 ʼ 標記 —— 入派三聲後它就是平/上/去，標記只會干擾合成。
 *    入聲的調值沿用《中原音韻》的派入：全濁入→陽平 ³⁵、次濁入/影母→去聲 ⁵¹、清入→上聲 ²¹。
 * 4. 異讀一律取前一讀（層次 0，白讀），一個音韻地位只出一個音節。
 *    例外三組取後一讀（共 4 字）：
 *      茂/莓（明母侯韻去）→ mɨw⁵¹（不取 maw）
 *      佛（文韻並母入）  → fu³⁵  （不取 fɔ）
 *      入（日母深攝入）  → ɻi⁵¹ （不取 ɻju）
 *    做法是把原方案這三條「層次 0」例外停用，讓它們落回常規規則。
 */

function 調整音韻地位() {
  function 調整(表達式, 調整屬性, 邊緣地位種類 = []) { if (is(表達式)) 音韻地位 = 音韻地位.調整(調整屬性, 邊緣地位種類); }
  // 輕唇化例外
  調整('明母 尤韻', { 等: '一', 類: null, 韻: '侯' });
  調整('明母 東韻', { 等: '一', 類: null });

  調整('云母 通攝 平聲', { 母: '匣' }, ['匣母三等']); // 熊
  if (!例外) return;

  // 流攝脣音入遇攝字
  調整('明母 侯韻 上聲', { 韻: '模' }); // 母某牡畝（謀戊在《廣韻》中涉及的字太多，不加入）
  /* 並母去聲也歸虞韻（使用者確認）：復/複 讀 fu，不再落進尤韻的 ʌw → waw */
  if (層次 === 0) 調整('尤韻 並母 平上去聲', { 韻: '虞' }); // 浮、婦阜負、復複（《中原音韻》僅魚模，元曲押魚模、尤侯）
  調整('尤韻 幫滂母 去聲', { 韻: '虞' }); // 富、副

  // 蟹攝二等入假攝字
  if (when([
    ['佳韻', [
      ['並母 上聲', true], // 罷
      ['見母 開口 平聲', true], // 佳
      ['溪母 合口 平聲', true], // 咼
      ['見匣母 合口 去聲', true], // 卦掛、畫
      [層次 === 1 && '疑母 開口 平聲', true], // 涯\崖
      [層次 === 1 && '生母 開口 去聲', true], // 洒\曬
    ]],
    ['夬韻 匣母 合口 去聲', true], // 話
  ])) 調整('蟹攝', { 韻: '麻' });

  調整('端母 蕭韻 上聲', { 母: '泥' }); // 鳥
  調整('生母 山韻 上聲', { 母: '初' }); // 産
  調整('書母 通攝 舒聲', { 母: '昌' }); // 舂
  調整('書母 鍾韻 入聲', { 母: '昌' }); // 束
  if (層次 === 0) 調整('書母 支韻 開口 去聲', { 母: '昌' }); // 翅\施
  調整('見母 蕭韻 平聲', { 母: '曉' }); // 梟鴞驍

  if (層次 === 0) {
    調整('果攝 開口 (定泥母 去聲 或 透母 平聲)', { 韻: '麻', 等: '二' }, ['端組類隔']); // 大那+他
    調整('端母 庚韻 上聲', { 韻: '麻' }, ['端組類隔']); // 打
  }
}

 *   全濁入 → 陽平 ³⁵｜次濁入・影母 → 去聲 ⁵¹｜清入 → 上聲 ²¹

       因為 ʼ 標記已經取消，不能再靠它認入聲。
       皆來韻入聲寫 ɛj，但前面帶 j 的（jaj，即梗二開見入）要保持 jaj，
       所以用負向先行斷言 (?<!j) 跳過它們 —— 否則「格」會被寫成 kjɛj。 */
    '音值（unt, 2021）': [
      ['(?<!j)aj', 皆來韻古入聲字韻基, is`入聲`],
      ['β', ''],
      ['jɨ', 'i'], ['ij', 'i'],
      ['jw', 'ɥ'],
      ['wɨ', 'u', !韻母.includes('ŋ')],
      ['ɥɨ', 'y', !韻母.includes('ŋ'), 'ɥi'],
      ['ʌw', 'waw', is脣音 && !is`入聲`, 'aw'],
      ['wʌ', 'ɔ', is脣音 || is銳音 && 韻母 === 'wʌ', 'wɔ'],
      ['ɥʌ', 'jɔ'],
      ['ɨ', 高元音開口呼], ['ə', 'ɹ̩', 韻母 === 'ə'], ['ɹ̩', 'ɻ̍', !聲母.includes('s')],
    ],
  };
  let 音节 = 聲母 + 韻母 + 聲調;
  轉換規則字典[顯示].forEach(規則 => {
    if (規則.length === 2 || 規則[2]) 音节 = 音节.replace(new RegExp(規則[0], 'g'), 規則[1]);
    else if (規則.length === 4) 音节 = 音节.replace(new RegExp(規則[0], 'g'), 規則[3]);
  });
  return 音节;
}
