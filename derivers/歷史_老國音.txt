/* 推導老國音 —— 純 IPA 版
 *
 * 改寫自 307587 的原方案（onp.js）。原方案以注音符號推導，最後再查一張表轉成國際音標，
 * 並且可以切換「注音符號 / 國際音標」兩種輸出。本版只輸出 IPA：
 *
 *   · 注音符號輸出分支整個去掉：規則直接寫成 IPA，不再經注音符號中轉
 *   · 只服務於注音輸出的設定項一併去掉（原「標注方式」「省略陰平調號」）
 *   · 調值固定用趙元任的五度數字（⁵⁵ ³⁵ ²¹⁴ ⁵¹ ⁵）：本項目合成器只認數字，
 *     ˉˊˇˋ˙ 那套調符會被當成音段丟掉，留着它等於沒有聲調
 *   · 介音固定寫 i u y；uən 固定寫 uən —— 這兩項原本的設定已刪
 *
 * 設定項只剩兩個：
 *   喉塞 —— 入聲是否寫出喉塞韻尾 ʔ
 *   匣上 —— 匣母上聲是否變去聲（匣母上聲有很多字不變去聲）
 *
 * 其餘原本可切換的寫法／音系分合項，按使用者要求一律固定寫死（默認值不變）：
 *   顎化     —— 牙喉音細音固定寫舌面前 tɕ tɕʰ ɕ，細音鼻音 ȵ（不再有不顎化／舌面中兩檔）
 *   鼻音與空韻 —— 固定寫 ȵ ɿ ʅ（另一套 ɲ ɹ̩ ɻ̍ 已去掉）
 *   iai      —— 固定寫 iai（不再可切 ie）
 *   區分官關／區分間堅／區分橘菊／區分國郭／區分楚處 —— 全部關閉（與原方案一致）：
 *     官關：山攝一等讀 uan（官 kuan ＝ 關 kuan）
 *     間堅：咸山攝二等開口牙喉音讀 iɛn（間 kiɛn ＝ 堅 kiɛn）
 *     橘菊：通攝三等入聲讀 yʔ（菊 tɕyʔ ＝ 橘 tɕyʔ）
 *     國郭：曾梗攝一二等合口入聲讀 uo（國 kuoʔ ＝ 郭 kuoʔ）
 *     楚處：莊組遇攝讀 u（初 tʂʰu ＝ 除 tʂʰu）
 *
 * 對拍：3804 個廣韻音韻地位（默認設定），與原方案「國際音標」模式的輸出逐字一致。
 * 改規則後請重跑對拍再提交。
 *
 * 音系：https://zh.wikiversity.org/zh-hant/老國音音素
 * 音變規則：https://zh.wikiversity.org/zh-hant/老國音與廣韻對比
 * 字音來源：https://github.com/baopaau/rime-bepemefeve/blob/main/bepemefeve.dict.yaml
 * 注意「僞老國音」：https://www.bilibili.com/read/cv17377530/
 *
 * 原 @author 307587
 */

/** @type { 音韻地位['屬於'] } */

/* 調值固定用趙元任的五度記法 —— 本項目合成器只認數字，別的寫法等於沒有聲調 */
const 聲調規則 = () => when([
  ['平聲', [['清音', '⁵⁵'], ['濁音', '³⁵']]],
  ['去聲 或 上聲 全濁', '⁵¹'],
  ['上聲', '²¹⁴'],
  ['入聲', '⁵'],
], '無聲調規則');

let 聲母 = 聲母規則();
let 韻母 = is`舒聲` ? 舒聲韻母規則() : 入聲韻母規則();
let 聲調 = 聲調規則();

if (!匣上變去 && is`匣母 上聲`) 聲調 = '²¹⁴';

/* ─────────────── 介音與韻母調整 ─────────────── */

/** 介音：i / u / y。ioŋ（撮口）與 oŋ（合口）的介音在 IPA 拼法裏不寫出來，要記住 */
const 介音 = () => {
  if (韻母 === 'ioŋ') return 'y';
  if (韻母 === 'oŋ') return 'u';
  return 'iuy'.includes(韻母[0]) ? 韻母[0] : '';
};

/*
 * 去掉介音之後的韻母。這幾張表就是原方案「把注音字串 slice(1) 再查對照表」的結果，
 * 例如 iɛn → an（不是 ɛn：丟掉 i 介音之後對應的韻是 an）、in → ən、ioŋ → oŋ。
 */
const 去i介音 = {
  i: 'əi', ia: 'a', io: 'o', ie: 'e', iai: 'ai', iau: 'au',
  iəu: 'əu', iɛn: 'an', in: 'ən', iaŋ: 'aŋ', iŋ: 'əŋ',
};
const 去u介音 = { ua: 'a', uo: 'o', uai: 'ai', uəi: 'əi', uan: 'an', uaŋ: 'aŋ' }; // uən、oŋ 不在此列
const 去y介音 = { y: 'u', yo: 'uo', ye: 'uo', yɛn: 'uan', yn: 'uən', ioŋ: 'oŋ' };

const 去介音 = () => ({ i: 去i介音, u: 去u介音, y: 去y介音 })[介音()]?.[韻母] ?? 韻母;
const 細音 = () => ['i', 'y'].includes(介音());

// 顎化：牙喉音 k kʰ x 在 i / y 介音前（固定舌面前：tɕ tɕʰ ɕ，見檔頭）
if (細音()) 聲母 = 顎化[聲母] || 聲母;
// 泥孃母細音（齊之韻開口除外）
if (細音() && 聲母 === 'n' && !is`齊之韻 開口`) 聲母 = 泥孃細鼻;
// 疑母細音脫鼻（規則本身不隨顎化程度變，只換符號：ŋ 或 ɲ）
if (介音() === 'i' && 聲母 === 'ŋ') 聲母 = is`(齊之尤蒸陽韻 非 合口) 或 (侵仙先元庚韻 入聲)` ? 疑母細鼻 : '';
if (介音() === 'y' && 聲母 === 'ŋ') 聲母 = '';

// 捲舌音：只去 i / y 介音，u 介音保留（「衰」ʂuai）
if (['tʂ', 'tʂʰ', 'ʂ', 'ɻ'].includes(聲母)) {
  if (韻母 === 'i') 韻母 = 'ɻ̍';            // 止攝知莊章組的空韻
  else if (細音()) 韻母 = 去介音();
}

// 脣音：y 介音 → u（y → u，yo/ye → uo，yɛn → uan，yn → uən，ioŋ → oŋ）
if (is`脣音` && 介音() === 'y') 韻母 = 去介音();

// 脣音：u 介音在非 uən、oŋ 前丟失（puo → po，但 puən 保留）
if (['p', 'pʰ', 'm', 'f'].includes(聲母) && 介音() === 'u' && 去u介音[韻母]) 韻母 = 去u介音[韻母];
// f / ʋ 後的 i 介音丟失，只剩下 i 時讀 əi（「微」ʋəi）
if (['f', 'ʋ'].includes(聲母) && 介音() === 'i') 韻母 = 去介音();

if (聲母 === 'ŋ' && 介音() === 'u' && !is`一等 山宕通攝 入聲`) 聲母 = ''; // 疑母合口

/* ─────────────── 寫法 ─────────────── */

{
  if (聲母 === 'ɲ') 聲母 = 'ȵ';
  if (韻母 === 'ɹ̩') 韻母 = 'ɿ';
  if (韻母 === 'ɻ̍') 韻母 = 'ʅ';
}

if (喉塞標尾 && is`入聲`) 韻母 += 'ʔ'; // 喉塞音韻尾

return 聲母 + 韻母 + 聲調;
