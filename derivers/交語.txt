function 聲母規則() {
  if (is('幫母')) {
    if(['不'].includes(字頭)&&is('入聲'))return 'ɓ';
    return is('東韻 三等 或 鍾微廢文元凡虞尤陽韻')?'f':is('A類')?'t':'ɓ';
  }
  if (is('滂母')) {
    return is('東韻 三等 或 鍾微廢文元凡虞尤陽韻')?'f':is('A類')?'tʰ':擦化?'f':'pʰ';
  }
  if (is('並母')) {
    return is('東韻 三等 或 鍾微廢文元凡虞尤陽韻')?'f':is('A類')?'t':'ɓ';
  }
  if (is('明母')) {
    return is('微廢文元凡虞陽韻')?'v':is('A類')?'z':'m';
  }

  if (is('端母')) return 'ɗ';
  if (is('透母')) return 'tʰ';
  if (is('定母')) return 'ɗ';
  if (is('泥母')) return 'n';
  if (is('來母 非 三等')) return 'l';

  if (is('知母')) return 'tʂ';
  if (is('徹母')) return 擦化?'ʂ':'tʂʰ';
  if (is('澄母')) return 'tʂ';
  if (is('孃母')) return 'n';
  if (is('來母')) return 'l';

  if (is('精母')) return 't';
  if (is('清母')) return 'tʰ';
  if (is('從母')) return 't';
  if (is('心母')) return 心邪母讀t?'t':'s';
  if (is('邪母')) return 心邪母讀t?'t':'s';

  if (is('莊母')) return 'tʂ';
  if (is('初母')) return 擦化?'ʂ':'tʂʰ';
  if (is('崇母')) return 'ʂ';
  if (is('生母')) return 'ʂ';
  if (is('俟母')) return 'ʂ';

  if (is('章母')) return 'tɕ';
  if (is('昌母')) return 擦化?'ɕ':'tɕʰ';
  if (is('常母')) return 書常船母讀th?'tʰ':'ɕ';
  if (is('書母')) return 書常船母讀th?'tʰ':'ɕ';
  if (is('船母')) return 書常船母讀th?'tʰ':'ɕ';
  if (is('日母')) return 'ȵ';

  if (is('見母')) return 'k';
  if (is('溪母')) return 擦化?'x':'kʰ';
  if (is('羣母')) return 'k';
  if (is('疑母')) return 'ŋ';

  if (is('曉母')) return 'h';
  if (is('匣母')) return 'h';
  if (is('影母')) return is('真韻 開口')?'ȵ':'';
  if (is('云母')) return 'h';
  if (is('以母')) return 'z';

  throw new Error('無聲母規則');
}

function 韻母規則() {
  // 通攝
  if (is('通攝 三等')) return 'uŋ';
  if (is('通攝')) return 'oŋ';

  // 江攝
  //if (is('江韻 莊組')) return 'ɔŋ';
  if (is('江韻')) return 'aŋ';

  // 止攝
  if (is('止攝 精組 開口')) return 'ɯ';
  if (is('止攝')) return 'i';


  // 遇攝
  if (is('魚韻')) return is('莊組')?'ɤ':'ɯ';
  if (is('虞韻')) return is('莊組')?'o':'u';
  if (is('模韻')) return 'o';

  // 蟹攝
  if (is('祭齊韻')) return 'e';
  if (is('廢韻')) return 'ə';
  if (is('泰韻')) return 'aj';
  if (is('佳皆夬韻')) return 'aj';
  if (is('灰韻')) return 'oj';
  if (is('咍韻')) return 'aj';

  // 臻攝
  if (is('元韻')) return is('幫組')?'an':'ien';
  if (is('臻攝 三等')) return is('莊組')?'ɐn':'ən';
  if (is('魂韻')) return 'on';
  if (is('痕韻')) return 'ən';

  // 山攝
  if (is('寒韻')) return 'an';
  if (is('刪山韻')) return 'an';
  if (is('仙先韻')) return 'ien';

  // 效攝
  if (is('蕭韻')) return 'iew';
  if (is('宵韻')) return 'iew';
  if (is('肴韻')) return 'aw';
  if (is('豪韻')) return 'aw';

  // 果攝
  if (is('歌韻 一等')) return 'a';
  if (is('歌韻 三等')) return is('幫組')?'a':'a';

  // 假攝
  if (is('麻韻 二等')) return 'a';
  if (is('麻韻 三等')) return 'a';
  if (is('麻韻 四等')) return 'a';

  // 宕攝
  if (is('陽韻 莊組')) return 'aŋ';
  if (is('陽韻 合口 非 云母')) return 'oŋ';
  if (is('陽韻')) return 'ɯɤŋ';
  if (is('唐韻')) return 'aŋ';

  // 梗攝
  //if (is('梗攝 端組 二等')) return 'a';
  if (is('梗攝 莊組')) return 'aiŋ';
  if (is('梗攝 影母 開口 舒聲')) return 'aiŋ';
  if (is('梗攝 二等')) return 'aiŋ';
  if (is('梗攝 三四等')) return 'iŋ';

  // 曾攝
  if (is('蒸韻')) return is('幫精莊組 舒聲')?'ɐŋ':'ɯŋ';
  if (is('登韻')) return 'ɐŋ';

  // 流攝
  if (is('幽韻')) return 'əw';
  if (is('尤韻 幫滂並母')) return 'u';
  if (is('尤韻')) return is('幫莊組')?'əw':'ɯw';
  if (is('侯韻')) return 'əw';

  // 深攝
  if (is('侵韻')) return is('莊組')?'ɐm':'əm';

  // 咸攝
  //if (is('談韻 舌齒音')) return 'am';
  if (is('覃談韻')) return 'am';
  if (is('凡韻')) return 'am';
  if (is('嚴鹽添韻')) return 'iem';
  if (is('咸銜韻')) return 'am';

  throw new Error('無韻母規則');
}

function 聲調規則() {
  if (!選項.拼音化) {
    if (is('全清 或 次清')) {
      if (is('平聲')) return '³³'; // 陰平
      if (is('上聲')) return '³¹³'; // 陰上
      if (is('去聲')) return '⁴⁵'; // 陰去
      if (is('入聲')) return '⁴⁵'; // 陰入
    }
    else {
      if (is('次濁 平聲')) return '³³'; 
      if (is('平聲')) return '³¹'; // 陽平
      if (is('全濁 上聲')) return 'ˀ²¹'; // 陽上
      if (is('上聲')) return 'ˀ³⁵'; // 陽上
      if (is('去聲')) return 'ˀ²¹'; // 陽去
      if (is('次濁 入聲')) return 'ˀ²¹'; // 陽入
      if (is('入聲')) return 'ˀ²¹'; // 陽入
    }
  }
  else
{
    if (is('全清 或 次清')) {
      if (is('平聲')) return ''; // 陰平
      if (is('上聲')) return '1'; // 陰上
      if (is('去聲')) return '2'; // 陰去
      if (is('入聲')) return '2'; // 陰入
    }
    else {
      if (is('次濁 平聲')) return ''; 
      if (is('平聲')) return '3'; // 陽平
      if (is('全濁 上聲')) return '5'; // 陽上
      if (is('上聲')) return '4'; // 陽上
      if (is('去聲')) return '5'; // 陽去
      if (is('次濁 入聲')) return '5'; // 陽入
      if (is('入聲')) return '5'; // 陽入
    }
  }
  throw new Error('無聲調規則');
}

  if(聲母==='k')
