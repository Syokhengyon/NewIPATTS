function 聲調規則() {

    }
  }

  throw new Error('無聲調規則');
}

/* 影、疑母 開口洪音：現代北京音兩者已混，故同一個開關決定怎麼寫。
   判準只用韻母首字符（不寫 i / y / u / ʊ）：i、y 起頭是細音（疑母細音已於上段併入 n），
   u、ʊ 起頭是合口性質；兩者都不屬於「開口洪音」。
   如此「侯」這類開合中立而實際讀開口的洪音能進來（偶 ŋəu、歐 ŋəu），
   而「模」讀 u、「東」讀 ʊŋ 這類實際是合口的會被排除（五 u、烏 u、翁 ʊŋ 不動）。 */
if (影疑洪音 === 'ŋ' && is`(影母 或 疑母)` && !'iuyʊ'.includes(韻母[0] || '')) 聲母 = 'ŋ';

else
{

if(韻母==='ə')韻母=韻母+'i';
韻母=韻母.replace('y','u');
韻母=韻母.replace('uɨn','uən');
韻母=韻母.replace('uɨʔ','uəʔ');

}

韻母=韻母.replace('uɔ','ɔ');

/* 臻深咸山曾梗 入聲（以及假攝）的主元音：按「攝 + 聲母組」分五欄定值。
 *
 *   欄位       默認   併入 ɛ   併入 ə   ←（不寫 ʔ 時固定按「併入 ɛ」那一欄，再去掉 ʔ）
 *   ① 臻深曾梗·知章日   ə      ɨ        ɨ
 *   ② 臻深曾梗·莊組     ə      ɛ        ə
 *   ③ 臻曾梗·其餘一二等 ə      ɛ        ə
 *   ④ 咸山·知章日       ɛ      ɛ        ə
 *   ⑤ 假攝·知章日       ɛ      ɛ        ə
 *   合口各自作 uə / uɛ / ʉ；只處理洪音（韻母不含 i / y），細音如 iɛʔ、yɛʔ 不動。
 */
{
  const 知章日 = is`知章組 或 日母`;
  const 莊 = is`莊組`;
  const 臻深曾梗 = is`臻攝 或 深攝 或 曾攝 或 梗攝`;
  const 咸山 = is`山攝 或 咸攝`;
  const 模式 = 寫入聲尾 ? 入聲主元音 : '併入 ɛ';
  const 表 = {
    '默認': { 1: 'ə', 2: 'ə', 3: 'ə', 4: 'ɛ', 5: 'ɛ' },
    '併入 ɛ': { 1: 'ɨ', 2: 'ɛ', 3: 'ɛ', 4: 'ɛ', 5: 'ɛ' },
    '併入 ə': { 1: 'ɨ', 2: 'ə', 3: 'ə', 4: 'ə', 5: 'ə' },
  }[模式] || null;
  if (表 && /^u?[əɛ]ʔ?$/.test(韻母)) {
    let 欄 = 0;
    if (臻深曾梗 && 知章日) 欄 = 1;
    else if (臻深曾梗 && 莊) 欄 = 2;
    else if (is`臻攝 或 曾攝 或 梗攝` && !知章日 && !莊) 欄 = 3;
    else if (咸山 && 知章日) 欄 = 4;
    else if (is`假攝` && 知章日) 欄 = 5;
    if (欄) {
      const 尾 = 韻母.endsWith('ʔ') ? 'ʔ' : '';
      const 合 = 韻母.startsWith('u');
      let v = 表[欄];
      if (合) v = v === 'ɨ' ? 'ʉ' : 'u' + v;
      韻母 = v + 尾;
    }
  }
}

/* 不寫入聲韻尾時：去掉 ʔ，順帶把 ʊ 併作 u（調值已在 聲調規則 裏改作 ³³） */
