function 聲母規則() {
  if (is('幫母')) {
    if(['不'].includes(字頭)&&is('入聲'))return 'p';
    return is('東韻 三等 或 鍾微廢文元凡虞尤陽韻')?'f':'p';
  }
  if (is('滂母')) {
    return is('東韻 三等 或 鍾微廢文元凡虞尤陽韻')?'f':'pʰ';
  }
  if (is('並母')) {
    return is('東韻 三等 或 鍾微廢文元凡虞尤陽韻')?'f':is('平聲')?'pʰ':'p';
  }
  if (is('明母')) {
    return 'm';
  }

  if (is('端母')) return 't';
  if (is('透母')) return 'tʰ';
  if (is('定母')) return is('平聲')?'tʰ':'t';
  if (is('泥母')) return 'n';
  if (is('來母 非 三等')) return 'l';

  if (is('知母')) return 'tʃ';
  if (is('徹母')) return 'tʃʰ';
  if (is('澄母')) return is('平聲')?'tʃʰ':'tʃ';
  if (is('孃母')) return 'n';
  if (is('來母')) return 'l';

  if (is('精母')) return 'ts';
  if (is('清母')) return 'tsʰ';
  if (is('從母')) return is('平聲')?'tsʰ':'ts';
  if (is('心母')) return 's';
  if (is('邪母')) return is('平聲')?'tsʰ':'ts';

  if (is('莊母')) return 'tʃ';
  if (is('初母')) return 'tʃʰ';
  if (is('崇母')) return is('平聲')?'tʃʰ':is('止攝')?'ʃ':'tʃ';
  if (is('生母')) return 'ʃ';
  if (is('俟母')) return 'ʃ';

  if (is('章母')) return 'tʃ';
  if (is('昌母')) return 'tʃʰ';
  if (is('常母')) return 'ʃ';
  if (is('書母')) return 'ʃ';
  if (is('船母')) return 'ʃ';
  if (is('日母')) return 'ɲ';

  if (is('見母')) return 'k';
  if (is('溪母')) return 溪母不擦化.includes(字頭)?'kʰ':'h';
  if (is('羣母')) return is('平聲')?'kʰ':'k';
  if (is('疑母')) return 'ŋ';

  if (is('曉母')) return 'h';
  if (is('匣母')) return is('開口 一二等 或 齊韻 開口')?'h':'';
  if (is('影母')) return '';
  if (is('云母')) return is('東韻 舒聲')?'h':'';
  if (is('以母')) return '';

  throw new Error('無聲母規則');
}

function 韻母規則() {
  // 通攝
  if (is('東冬鍾韻')) return 'ʊŋ';

  // 江攝
  if (is('江韻')) return 'ɔŋ';

  // 止攝
  if (is('支脂之微韻 開口')) return is('莊組')?'ɔj':'i';
  if (is('支脂之微韻 合口')) return is('莊組')?'uj':'ɐj';
  if (is('支脂之韻')) return 'i';
  if (is('微韻')) return 'i';


  // 遇攝
  if (is('魚韻')) return is('莊組')?'ɔ':'y';
  if (is('虞韻')) return is('幫莊組')?'u':'y';
  if (is('模韻')) return 'u';

  // 蟹攝
  if (is('齊韻')) return 'ɐj';
  if (is('祭韻')) return is('莊組')?'uj':'ɐj';
  if (is('泰灰韻 幫組')) return 'uj';
  if (is('泰韻')) return is('舌齒音 開口')?'aj':'ɔj';
  if (is('佳皆夬韻')&&蟹攝二等歸假攝字.includes(字頭)) return 'a';
  if (is('佳皆夬韻')) return 'aj';
  if (is('灰咍韻')) return 'ɔj';
  if (is('廢韻 開口')) return is('章組')?'ɔj':'i';
  if (is('廢韻')) return 'ɐj';

  // 臻攝
  if (is('真韻')) return is('莊組')?'ɐn':'ən';
  if (is('殷韻')) return 'ən';
  if (is('臻韻')) return 'ɐn';
  if (is('文韻')) return is('幫組')?'ɐn':'ən';
  if (is('元韻')) return is('幫組')?'an':'in';
  if (is('魂韻')) return 'ɐn';
  if (is('痕韻')) return 'ɐn';

  // 山攝
  if (is('寒韻 舌齒音')) return is('開口')?'an':'un';
  if (is('寒韻')) return is('開口')?'ɔn':'un';
  if (is('刪山韻')) return 'an';
  if (is('仙先韻')) return 'in';

  // 效攝
  if (is('蕭韻')) return 'iw';
  if (is('宵韻')) return 'iw';
  if (is('肴韻')) return 'aw';
  if (is('豪韻')) return 'ow';

  // 果攝
  if (is('歌韻 一等')&&果攝滯古.includes(字頭)) return 'a';
  if (is('歌韻 一等')) return 'ɔ';
  if (is('歌韻 三等')) return 'œ';

  // 假攝
  if (is('麻韻 二等')) return 'a';
  if (is('麻韻 三等')) return 'e';
  if (is('麻韻 四等')) return 'e';

  // 宕攝
  if (is('陽韻 幫組')) return 'ɔŋ';
  if (is('陽韻')) return is('莊組')?'ɔŋ':'œŋ';
  if (is('唐韻')) return 'ɔŋ';

  // 梗攝
  if (is('庚耕韻 端組')) return 'a';
  if (is('庚耕韻 二等')) return 'aŋ';
  if (is('庚耕韻 莊組')) return 'aŋ';
  if (is('庚韻 三等')) return 'iŋ';
  if (is('青韻')) return 'iŋ';
  if (is('清韻')) return 'iŋ';


  // 曾攝
  if (is('蒸韻')) return is('莊組')?'ɐŋ':'iŋ';
  if (is('登韻')) return 'ɐŋ';

  // 流攝
  if (is('尤侯韻 明母')&&流攝明母歸遇攝字.includes(字頭)) return 'u';
  if (is('尤侯韻 幫滂並母')&&流攝幫滂並母歸遇攝字.includes(字頭)) return 'u';
  if (is('尤韻')) return is('幫莊組')?'ɐw':'əw';
  if (is('幽韻')) return is('幫組')?'iw':'əw';
  if (is('侯韻')) return 'ɐw';

  // 深攝
  if (is('侵韻')) return is('莊組')?'ɐm':'əm';

  // 咸攝
  if (is('覃談韻 舌齒音')) return 'am';
  if (is('覃談韻')) return 'ɔm';
  if (is('凡韻')) return 'am';
  if (is('嚴鹽添韻')) return 'im';
  if (is('咸銜韻')) return 'am';

  throw new Error('無韻母規則');
}

function 聲調規則() {
  if (1) {
    if (is('全清 或 次清')) {
      if (is('平聲')) return '⁵⁵'; // 陰平
      if (is('上聲')) return '³⁵'; // 陰上
      if (is('去聲')) return '³³'; // 陰去
      if (is('入聲')) return '⁵'; // 陰入
    }
    else {
      if (is('平聲')) return '³¹'; // 陽平
      if (is('全濁 上聲')) return '²²'; // 陽上
      if (is('上聲')) return '²³'; // 陽上
      if (is('去聲')) return '²²'; // 陽去
      if (is('次濁 入聲')) return '²'; // 陽入
      if (is('入聲')) return '²'; // 陽入
    }
  }

  throw new Error('無聲調規則');
}
