const is = (x) => 音韻地位.屬於(x);

function 聲母規則() {
  if (is('幫母')) {
    if(['不'].includes(字頭)&&is('入聲'))return 'p';
    return is('東韻 三等 或 鍾微廢文元凡虞尤陽韻')?'f':'p';
  }
  if (is('滂母')) {
    return is('東韻 三等 或 鍾微廢文元凡虞尤陽韻')?'f':'pʰ';
  }
  if (is('並母')) {
    return is('東韻 三等 或 鍾微廢文元凡虞尤陽韻')?'v':is('平聲')?'bʱ':'b';
  }
  if (is('明母')) {
    return is('微廢文元凡虞陽韻')?'ʋ':'m';
  }

  if (is('端母')) return 't';
  if (is('透母')) return 'tʰ';
  if (is('定母')) return is('平聲')?'dʱ':'d';
  if (is('泥母')) return 'n';
  if (is('來母')) return 'l';

  if (is('知母')) return 'tɹ';
  if (is('徹母')) return 'tʰɹ';
  if (is('澄母')) return is('平聲')?'dʱɹ':'dɹ';
  if (is('孃母')) return 'n';

  if (is('精母')) return 'ts';
  if (is('清母')) return 'tsʰ';
  if (is('從母')) return is('平聲')?'dzʱ':'dz';
  if (is('心母')) return 's';
  if (is('邪母')) return 'z';

  if (is('莊母')) return 'tʃ';
  if (is('初母')) return 'tʃʰ';
  if (is('崇母')) return is('平聲')?'dʒʱ':is('止攝')?'ʒ':'dʒ';
  if (is('生母')) return 'ʃ';
  if (is('俟母')) return 'ʒ';

  if (is('章母')) return 'tʃ';
  if (is('昌母')) return 'tʃʰ';
  if (is('常母')) return is('平聲')?'dʒʱ':'ʒ';
  if (is('書母')) return 'ʃ';
  if (is('船母')) return 'ʒ';
  if (is('日母')) return 'ɹ';

  if (is('見母')) return 'k';
  if (is('溪母')) return 'kʰ';
  if (is('羣母')) return is('平聲')?'ɡʱ':'ɡ'
  if (is('疑母')) return 'ŋ';

  if (is('曉母')) return 'x';
  if (is('匣母')) return 'ɣ';
  if (is('影母')) return 'ʔ';
  if (is('云母')) return '';
  if (is('以母')) return 'j';

  throw new Error('無聲母規則');
}

function 韻母規則() {
  // 通攝
  if (is('東冬韻 一等')) return 'ʊŋ';
  if (is('東鍾韻 三等')) return is('幫莊組')?'ʊŋ':'iʊŋ';
  if (is('東鍾韻 三等')) return is('幫莊組')?'ʊŋ':'yʊŋ';

  // 江攝
  if (is('江韻 知莊組')) return 'uaŋ';
  if (is('江韻 來母')) return 'uaŋ';
  if (is('江韻 牙喉音')) return is('影母')?'uaŋ':'aŋ';
  if (is('江韻')) return 'aŋ';

  // 止攝
  if (is('支脂之微韻 開口')) return is('精莊組')?'ɨ':'i';
  if (is('支脂之微韻 合口')) return 'yi';
  if (is('支脂之韻')) return 'i';
  if (is('微韻')) return 'i';


  // 遇攝
  if (is('魚韻')) return is('莊組')?'ɯ':'iɯ';
  if (is('虞韻')) return is('幫莊組')?'u':'yu';
  if (is('模韻')) return 'u';

  // 蟹攝
  if (is('齊韻')) return is('合口')?'yi':'i';
  if (is('祭韻')) return is('合口')?'yi':'i';
  if (is('泰韻 合口')) return is('見溪疑母')?'uɑj':'uɨi';
  if (is('泰韻')) return is('銳音')?'aj':'ɑj';
  if (is('佳皆夬韻')) return (['咼(呙)扠哇洼絓畫(画)罣罷(罢)', '佳鼃娃卦涯', '灑蝸話(话)掛挂查叉杈衩'].join('').includes('字頭'))?'a':'aj';
  if (is('灰韻')) return 'uɨi';
  if (is('咍韻')) return 'ɑj';
  if (is('廢韻 開口')) return 'i';
  if (is('廢韻')) return is('合口')?'yi':'ui';

  // 臻攝
  if (is('真韻')) return is('莊組')?'ɨn':'in';
  if (is('殷韻')) return 'in';
  if (is('臻韻')) return 'ɨn';
  if (is('文韻')) return is('幫組')?'un':'in';
  if (is('元韻 合口')) return 'yen';
  if (is('元韻')) return is('幫組')?'an':'ien';
  if (is('魂韻')) return 'un';
  if (is('痕韻')) return 'ɨn';

  // 山攝
  if (is('寒韻')) return is('銳音 開口')?'an':'ɑn';
  if (is('刪山韻')) return 'an';
  if (is('仙先韻')) return is('合口')?'yen':'ien';

  // 效攝
  if (is('蕭韻')) return 'iew';
  if (is('宵韻')) return 'iew';
  if (is('肴韻')) return 'aw';
  if (is('豪韻')) return 'ɑw';

  // 果攝
  if (is('歌韻 一等')) return 'ɑ';
  if (is('歌韻 三等 合口')) return 'yɑ';
  if (is('歌韻 三等')) return is('幫組')?'a':'iɑ';

  // 假攝
  if (is('麻韻 二等')) return 'a';
  if (is('麻韻 三等')) return 'ia';
  if (is('麻韻 四等')) return 'ia';

  // 宕攝
  if (is('陽韻 幫組')) return 'ɑŋ';
  if (is('陽韻 合口')) return 'yɑŋ';
  if (is('陽韻')) return is('莊組')?'ɑŋ':'iɑŋ';
  if (is('唐韻')) return 'ɑŋ';

  // 梗攝
  if (is('庚耕韻 二等')) return 'ɪŋ';
  if (is('庚耕韻 莊組')) return 'ɪŋ';
  if (is('庚韻 三等')) return 'iŋ';
  if (is('青韻')) return 'iŋ';
  if (is('清韻')) return 'iŋ';


  // 曾攝
  if (is('蒸韻')) return is('莊組')?'ɪŋ':'iŋ';
  if (is('登韻')) return is('合口')?'uɨŋ':'ɨŋ';

  // 流攝
  if (is('尤韻 幫滂並母'))return(['浮桴蜉䳕抔掊裒婄箁䍌蔀捊', '不否缶剖紑鴀缹秠'].join('').includes(字頭))?'ɨw':'u'
  if (is('尤韻')) return is('幫莊組')?'ɨw':'iw';
  if (is('幽韻')) return 'iw';
  if (is('侯韻 幫組')) return (['母拇', '戊姆呣'].join('').includes(字頭))?'u':'ɨw';
  if (is('侯韻')) return 'ɨw';

  // 深攝
  if (is('侵韻')) return is('莊組')?'ɨm':'im';

  // 咸攝
  if (is('覃韻')) return 'ɑm';
  if (is('談韻')) return is('銳音')?'am':'ɑm';
  if (is('凡韻')) return 'am';
  if (is('嚴鹽添韻')) return 'iem';
  if (is('咸銜韻')) return 'am';

  throw new Error('無韻母規則');
}

function 聲調規則() {
  if (1) {
    if (is('全清 或 次清')) {
      if (is('平聲')) return '⁴²'; // 陰平
      if (is('上聲')) return '⁵⁵'; // 陰上
      if (is('去聲')) return '³⁴'; // 陰去
      if (is('入聲')) return '⁴'; // 陰入
    }
    else {
      if (is('平聲')) return '²¹'; // 陽平
      if (is('全濁 上聲')) return '²⁴'; // 陽上
      if (is('上聲')) return '⁵⁵'; // 陽上
      if (is('去聲')) return '²⁴'; // 陽去
      //if (is('次濁 入聲')) return '³'; // 陽入
      if (is('入聲')) return '²²'; // 陽入
    }
  }

  throw new Error('無聲調規則');
}

let 聲母 = 聲母規則();
let 韻母 = 韻母規則();
let 聲調 = 聲調規則();

if (is('入聲')) {
if (韻母.endsWith('m')) 韻母 = 韻母.slice(0, -1) + 'p';
  else if (韻母.endsWith('n')) 韻母 = 韻母.slice(0, -1) + 't';
  else if (韻母.endsWith('ŋ')) 韻母 = 韻母.slice(0, -1) + 'k';

}

if (is('合口')) {
  韻母='u'+韻母;
韻母=韻母.replace('ui','yi');
韻母=韻母.replace('ww','w');
韻母=韻母.replace('uu','u');
韻母=韻母.replace('uy','y');
}
韻母=韻母.replace('uɨi','ui');


if(is('莊組'))
{
韻母=韻母.replace('yi','ui');
if(is('非 止攝'))韻母=韻母.replace('i','ɨ');
韻母=韻母.replace('ie','a');
韻母=韻母.replace('ye','ua');
}
return 聲母 + 韻母 + 聲調;
