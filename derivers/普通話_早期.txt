const 流攝幫滂並母未歸遇攝字 = ['浮桴蜉䳕抔掊裒婄箁䍌蔀捊', '不否缶剖紑鴀缹秠'].join('');
const 流攝明母歸遇攝字 = ['母拇', '戊姆呣'].join('');

function 聲母規則() {
  if (is('幫母')) {
    return is('東韻 三等 或 鍾微廢文元凡虞尤陽韻')?'f':'p';
  }
  if (is('滂母')) {
    return is('東韻 三等 或 鍾微廢文元凡虞尤陽韻')?'f':'pʰ';
  }
  if (is('並母')) {
    return is('東韻 三等 或 鍾微廢文元凡虞尤陽韻')?'fʱ':is('平聲')?'bʱ':'b';
  }
  if (is('明母')) {
    return is('微廢文元凡虞陽韻')?'v':'m';
  }

  if (is('端母')) return 't';
  if (is('透母')) return 'tʰ';
  if (is('定母')) return is('平聲')?'dʱ':'d';
  if (is('泥母')) return 'n';
  if (is('來母 非 三等')) return 'l';

  if (is('知母')) return 'tʃ';
  if (is('徹母')) return 'tʃʰ';
  if (is('澄母')) return is('平聲')?'dʒʱ':'dʒ';
  if (is('孃母')) return 'n';
  if (is('來母')) return 'l';

  if (is('精母')) return 'ts';
  if (is('清母')) return 'tsʰ';
  if (is('從母')) return is('平聲')?'dzʱ':'dz';
  if (is('心母')) return 's';
  if (is('邪母')) return 'sʱ';

  if (is('莊母')) return 'tʂ';
  if (is('初母')) return 'tʂʰ';
  if (is('崇母')) return is('止攝')?'ʂʱ':is('平聲')?'dʐʱ':'dʐ';
  if (is('生母')) return 'ʂ';
  if (is('俟母')) return 'ʂʱ';

  if (is('章母')) return 'tɕ';
  if (is('昌母')) return 'tɕʰ';
  if (is('常母')) return is('陽聲韻 平聲')?'dʑʱ':'ɕʱ';
  if (is('書母')) return 'ɕ';
  if (is('船母')) return 'ɕʱ';
  if (is('日母')) return 'ʒ';

  if (is('見母')) return 'k';
  if (is('溪母')) return 'kʰ';
  if (is('羣母')) return is('平聲')?'ɡʱ':'ɡ';
  if (is('疑母')) return 'ŋ';

  if (is('曉母')) return 'x';
  if (is('匣母')) return 'ɣ';
  if (is('影母')) return 'ʔ';
  if (is('云母')) return is('東韻 舒聲')?'ɣ':'';
  if (is('以母')) return '';

  throw new Error('無聲母規則');
}

function 韻母規則() {
  // 通攝
  if (is('東冬韻 一等')) return 'ʊŋ';
  if (is('東鍾韻 三等')) return is('幫莊組')?'ʊŋ':'jʊŋ';
  //if (is('東鍾韻 三等')) return is('牙喉音')?'jʊŋ':'ʊŋ';

  // 江攝
  if (is('江韻 牙喉音 非 影母')) return 'jaŋ';
  if (is('江韻')) return is('知莊組 或 影母')?'waŋ':'aŋ';

  // 止攝
  if (is('支脂之微韻 開口 章組')) return 'jɨ';
  if (is('支脂之微韻 開口')) return is('精莊組 或 日母')?'ɨ':'ji';
  if (is('支脂之微韻 合口')) return is('知章組 或 日母')?'ɥi':'wəj';
  if (is('支脂之韻')) return 'ji';
  if (is('微韻')) return 'əj';


  // 遇攝
  if (is('魚韻')) return is('莊組')?'u':'ɥu';
  if (is('虞韻')) return  is('幫莊組')?'u':'ɥu';
  if (is('模韻')) return 'u';

  // 蟹攝
  if (is('齊韻')) return is('合口')?'wəj':'ji';
  if (is('祭韻 知章組 合口')) return 'ɥi'
  if (is('祭韻 以母 合口')) return 'ɥi'
  if (is('祭韻 莊組 開口')) return 'ɨ'
  if (is('祭韻')) return is('合口')?'wəj':'ji';
  if (is('泰韻 疑母 合口')) return 'waj';
  if (is('泰韻')) return is('開口')?'aj':is('幫組')?'əj':'wəj';
  if (is('佳皆夬韻')&&蟹攝二等歸假攝字.includes(字頭)) return 'a';
  if (is('佳皆夬韻')) return 'aj';
  if (is('灰韻')) return is('合口')?'wəj':'əj';
  if (is('咍韻')) return 'aj';
  if (is('廢韻 開口')) return is('章組')?'jaj':'ji';
  if (is('廢韻')) return is('合口')?'wəi':'əj';

  // 臻攝
  if (is('真韻')) return is('莊組')?'ən':'jən';
  if (is('殷韻')) return 'jən';
  if (is('臻韻')) return 'ən';
  if (is('文韻')) return is('幫組')?'ən':'jən';
  if (is('元韻')) return is('幫莊組')?'an':'jɛn';
  if (is('魂韻')) return is('合口')?'wən':'ən';
  if (is('痕韻')) return 'ən';

  // 山攝
  if (is('寒韻')) return is('開口')?'an':'wɔn';
  if (is('刪山韻')) return 'an';
  if (is('仙韻')) return 'jɛn';
  if (is('先韻')) return is('莊組')?'an':'jɛn'

  // 效攝
  if (is('蕭韻')) return 'jɛw';
  if (is('宵韻')) return 'jɛw';
  if (is('肴韻')) return 'aw';
  if (is('豪韻')) return 'aw';

  // 果攝
  if (is('歌韻 一等')&&果攝滯古.includes(字頭)) return 'a';
  if (is('歌韻 一等')) return is('牙喉音 開口')?'ɔ':'wɔ';
  if (is('歌韻 三等')) return is('幫組')?'a':'jɛ';

  // 假攝
  if (is('麻韻 二等')) return 'a';
  if (is('麻韻 三等')) return 'jɛ';
  if (is('麻韻 四等')) return 'jɛ';

  // 宕攝
  if (is('陽韻 幫組')) return 'aŋ';
  if (is('陽韻 合口')) return is('入聲')?'jaŋ':'waŋ';
  if (is('陽韻')) return is('莊組')?'waŋ':'jaŋ';
  if (is('唐韻')) return 'aŋ';

  // 梗攝
  if (is('庚耕韻 二等 牙喉音 開口 非 入聲')) return 'jəŋ';
  if (is('庚耕韻 端組')) return 'a';
  if (is('庚耕韻 二等')) return 'əŋ';
  if (is('庚韻 莊組')) return 'əŋ';
  if (is('庚韻 三等')) return 'jəŋ';
  if (is('青韻')) return 'jəŋ';
  if (is('清韻')) return 'jəŋ';


  // 曾攝
  if (is('蒸韻')) return is('莊組')?'əŋ':'jəŋ';
  if (is('登韻')) return is('合口')?'wəŋ':'əŋ';

  // 流攝
  if (is('尤侯韻 明母')&&流攝明母歸遇攝字.includes(字頭)) return 'u';
  if (is('尤侯韻 明母')) return 'əw';
  if (is('尤侯韻 幫滂並母')&&流攝幫滂並母未歸遇攝字.includes(字頭)) return 'əw';
  if (is('尤侯韻 幫滂並母')) return 'u';
  if (is('尤韻')) return is('莊組')?'əw':'jəw';
  if (is('幽韻')) return 'jəw';
  if (is('侯韻')) return 'əw';

  // 深攝
  if (is('侵韻')) return is('莊組')?'əm':'jəm';

  // 咸攝
  if (is('覃韻')) return 'am';
  if (is('談韻')) return 'am';
  if (is('凡韻')) return 'am';
  if (is('嚴鹽添韻')) return is('莊組')?'am':'jɛm';
  if (is('咸銜韻')) return 'am';

  throw new Error('無韻母規則');
}

function 聲調規則() {
  if (1) {
    if (is('全清 或 次清')) {
      if (is('平聲')) return '⁵⁵'; // 陰平
      if (is('上聲')) return '⁵³'; // 陰上
      if (is('去聲')) return '²⁴'; // 陰去
      if (is('入聲')) return '⁵'; // 陰入
    }
    else {
      if (is('平聲')) return '²¹²'; // 陽平
      if (is('全濁 上聲')) return '³³'; // 陽上
      if (is('上聲')) return '⁵³'; // 陽上
      if (is('去聲')) return '³³'; // 陽去
      if (is('次濁 入聲')) return '²'; // 陽入
      if (is('入聲')) return '²'; // 陽入
    }
  }

  throw new Error('無聲調規則');
}

let 聲母 = 聲母規則();
let 韻母 = 韻母規則();
let 聲調 = 聲調規則();

if (is('入聲')) {
if (韻母.endsWith('m')) 韻母 = 韻母.slice(0, -1) + 'p';
  else if (韻母.endsWith('n')) 韻母 = 韻母.slice(0, -1) + 't';
  else if (韻母.endsWith('ŋ')) 韻母 = 韻母.slice(0, -1) + 'k';
}

if (is('合口')) {
  韻母='w'+韻母;
韻母=韻母.replace('wj','ɥ');
韻母=韻母.replace('ww','w');
韻母=韻母.replace('wu','u');
韻母=韻母.replace('wɥ','ɥ');
}

if(is('牙喉音 開口 二等 非 梗江攝'))
{
  韻母='j'+韻母;
  if(is('蟹攝 影母'))
  {韻母=韻母.replace(/^j/,'');}
}

if(is('宕江攝 入聲'))
{
韻母=韻母.replace('a','ɔ');
韻母=韻母.replace('ɥɔ','jɔ');
}
if(is('寒覃談韻 牙喉音 入聲'))
{
韻母=韻母.replace('a','ɔ');
}


if(is('知組 或 日母'))
{
if(韻母[0]==='j'||韻母[0]==='ɥ'||韻母[0]==='w')
{聲母=聲母.replace('ʃ','ɕ');
聲母=聲母.replace('ʒ','ʑ');}
else
{
聲母=聲母.replace('ʃ','ʂ');
聲母=聲母.replace('ʒ','ʐ');
}
}
/* ── 寫法統一 ──
   ① 介音一律寫 i / u / y（原稿寫 j / w / ɥ）；
   ② 韻尾 j / w 一律寫 i / u（原稿寫 j / w）；
   ③ 臻深曾梗攝的 jə / ɥə 併作 i / yi（真 jən→in、侵 jəm→im、蒸 jəŋ→iŋ、真合 ɥən→yin…）。
   ③ 要先做，否則 ① 之後就分不出「介音 j」和「ə 音首」了。 */
if (is`臻攝 或 深攝 或 曾攝 或 梗攝`) {
  韻母 = 韻母.replace(/jə/g, 'i').replace(/ɥə/g, 'yi');
}
韻母 = 韻母.replace(/^j/, 'i').replace(/^w/, 'u').replace(/^ɥ/, 'y');  // 介音
韻母 = 韻母.replace(/j$/, 'i').replace(/w$/, 'u');                     // 韻尾
/* 介音與韻腹同音時合併（ji→i、wu→u、yy→y） */
韻母 = 韻母.replace(/ii/g, 'i').replace(/uu/g, 'u').replace(/yy/g, 'y');

return 聲母 + 韻母 + 聲調;
