const is = (x) => 音韻地位.屬於(x);

function 聲母規則() {
  if (is('幫母')) {
    if (is('東韻 三等 或 鍾微虞廢文元陽尤凡韻')) return 'f';
    return 'p';
  }
  if (is('滂母')) {
    if (is('東韻 三等 或 鍾微虞廢文元陽尤凡韻')) return 'f';
    return 'pʰ';
  }
  if (is('並母')) {
if (is('東韻 三等 或 鍾微虞廢文元陽尤凡韻')) return 'v';return 'b';    
  }
  if (is('明母')) 
  {if (is('鍾微虞廢文元陽凡韻')) return 'v';return 'm';}

  if (is('端母')) return 't';
  if (is('透母')) return 'tʰ';
  if (is('定母')) return 'd';
  if (is('泥孃母')) return 'n';
  if (is('來母')) return 'l';

  if (is('精母')) return 'ts';
  if (is('清母')) return 'tsʰ';
  if (is('從母')) return 'z';
  if (is('心母')) return 's';
  if (is('邪母')) return 'z';

  if (is('江韻'))
    {
    if (is('知莊母')) return 'tɕ';
    if(is('徹初母')) return 'tɕʰ';
    if (is('澄母')) return 'dʑ';
    if(is('生母')) return 'ɕ';
    if(is('崇俟母')) return 'ʑ';
    }
  
  if (is('莊母 或 知母 二等')) return 'ts';
  if (is('初母 或 徹母 二等'))  return 'tsʰ';
  if (is('澄母 二等')) return 'dz';
  if (is('生母')) return 's' ;
  if (is('崇俟母')) return 'z' ;
  
  if (is('魚韻'))
    {
    if (is('知章母')) return 'ts';
    if(is('徹昌母')) return 'tsʰ';
    if(is('澄母')) return 'dz';
    if(is('書母')) return 's';
    if(is('常船母')) return 'z';
    }
  
    if (is('麻陽韻 或 止攝 開口'))
    {
    if (is('章母')) return 'ts';
    if(is('昌母')) return 'tsʰ';
    if(is('書母')) return 's';
    if(is('常船母')) return 'z';
    }
  
  if (is('知章母')) return 'tɕ';
  if (is('徹昌母'))  return 'tɕʰ';
  if (is('澄母')) return 'dʑ';
  if (is('常船母')) return 'ʑ' ;
  if (is('書母')) return  'ɕ';
  if (is('日母')) return 'ʑ';

  if (is('見母')) return 'k';
  if (is('溪母')) return 'kʰ'; // 多數擦化
  if (is('羣母')) return 'ɡ';
  if (is('疑母')) {if(is('三四等 非 魚韻')) return 'ȵ';else {return 'ŋ';}}
  
  if (is('曉母')) return 'h';
  if (is('匣云以母')) return 'ɦ';
  if (is('影母')) return 'ʔ';
  
  throw new Error('無聲母規則');
}

function 韻母規則() {
  // 通攝
  if (is('東冬韻 一等')) return 'ʊŋ';
  if (is('東鍾韻 幫莊組 三等')) return 'ʊŋ';
  if (is('東韻 三等')) return 'iʊŋ';
  if (is('鍾韻 三等')) return 'yɔŋ';
  // 江攝
  if (is('江韻 知莊組')) return 'yɔŋ';
  if (is('江韻')) return 'ɔŋ';

  // 止攝
  if (is('支脂之微韻 幫組')) return 'i';
  if (is('支脂之微韻 精組 開口')) return 'ɿ';
  if (is('支脂之微韻 章莊組 開口')) return 'ɿ';
  if (is('支脂之微韻 莊組 合口')) return 'ei';
  if (is('支脂之微韻 開口')) return 'i'; 
  if (is('支脂之微韻 合口')) return 'y';


  // 遇攝
  if (is('魚韻 莊組')) return 'ou';
  if (is('魚韻 牙音 或 魚韻 曉母')) return 'ei';
  if (is('魚韻 知章組 非 孃母')) return 'ɿ';
  if (is('魚韻 精組')) return 'i';
  if (is('虞韻 幫莊組')) return 'u';
  if (is('魚虞韻')) return 'y';
  if (is('模韻')) return 'u';

  // 蟹攝
  if (is('廢韻 幫組')) return 'i';
  if (is('祭齊廢韻 幫組')) return 'i';
  if (is('祭齊廢韻 開口')) return 'i';
  if (is('祭齊廢韻 合口')) return is("牙喉音")?'uei':'y';
  if (is('泰灰韻 幫組')) return 'ei';
  if (is('泰韻 開口 端組')) return 'a';
  if (is('泰韻 開口 來母')) return 'a';
  if (is('泰韻 開口 牙喉音 或 泰韻 開口 精組')) return 'ei';
  if (is('泰韻 合口 舌齒音')) return 'ei';
  if (is('泰韻 合口 牙喉音')) return 'uei';
  if (is('夬韻 合口 喉音')) return 'uo';
  if (is('佳皆夬韻 合口')) return 'ua';
  if (is('佳皆夬韻')) return 'a';
  if (is('灰韻 舌齒音')) return 'ei';
  if (is('灰韻 牙喉音')) return 'uei';
  if (is('咍韻')) return 'ei';
  //if (is('廢韻')) return 'ai';

  // 臻攝
  if (is('真韻 幫組')) return 'iŋ';
  if (is('文韻 幫組')) return 'əŋ';
  if (is('真臻韻 莊組 開口')) return 'əŋ';
  if (is('真殷韻 開口')) return 'iŋ';
  if (is('真臻韻 莊組 合口')) return 'uəŋ';
  if (is('真文韻 合口')) return 'yŋ';
  if (is('元韻 幫組')) return 'ɛn';
  if (is('元韻 開口 莊組')) return 'ɛn';
  if (is('元韻 合口 莊組')) return 'øn';
  if (is('元韻 開口')) return 'ien';
  if (is('元韻 合口')) return 'yøn';
  if (is('魂韻 牙喉音')) return 'uəŋ';
  if (is('魂韻')) return 'əŋ';
  if (is('痕韻')) return 'əŋ';

  // 山攝
  if (is('寒韻 幫組')) return 'œn';
  if (is('寒韻 開口 舌齒音')) return 'ɛn';
  if (is('寒韻 開口 牙喉音')) return 'œn';
  if (is('寒韻 合口 舌齒音')) return 'œn';
  if (is('寒韻 合口 牙喉音')) return 'uœn';
  if (is('刪山韻 合口 舌齒音')) return 'œn';
  if (is('刪山韻 合口')) return 'uɛn';
  if (is('刪山韻')) return 'ɛn';
  if (is('仙先韻 幫組')) return 'ien';
  if (is('仙先韻 開口 莊組')) return 'ɛn';
  if (is('仙先韻 合口 莊組')) return 'œn';
  if (is('仙先韻 開口')) return 'ien';
  if (is('仙先韻 合口')) return 'yøn';

  // 效攝
  if (is('蕭宵韻')) return 'ieu';
  if (is('肴韻')) return 'au';
  if (is('豪韻')) return 'au';

  // 果攝
  if (is('歌韻 幫組 一等')) return 'ou';
  if (is('歌韻 開口 一等')) return 'ou';
  
  if (is('歌韻 合口 一等 舌齒音')) return 'ou';
  if (is('歌韻 合口 一等 牙喉音')) return 'ou';
  
  if (is('歌韻 合口 三等')) return 'ya';
  if (is('歌韻 三等')) return 'ia';
  // 假攝
  if (is('麻韻 幫組 二等')) return 'o';
  if (is('麻韻 開口 二等')) return 'o';
  if (is('麻韻 合口 二等')) return 'uo';
  if (is('麻韻 章組')) return 'o';
  if (is('麻韻 三四等')) return 'ia';

  // 宕攝
  if (is('唐陽韻 幫莊組'))return 'ɔŋ';
  if (is('陽韻 章組'))return 'ɔŋ';
  //if (is('陽韻 開口 莊組')) return 'ɔŋ';
  if (is('陽韻 開口')) return 'iaŋ';
  if (is('陽韻 合口')) return 'yɔŋ';

  
  if (is('唐韻 開口')) return 'ɔŋ';
  if (is('唐韻 合口')) return 'uɔŋ';
  
  // 梗攝
  if (is('庚耕韻 二等 合口')) return 'uaŋ';
  if (is('庚耕韻 二等')) return 'aŋ'; 
  if (is('庚韻 三等 莊組')) return 'aŋ'; 
  if (is('庚清韻 三等 章組 入聲')) return 'iaʔ';
  if (is('庚韻 三等 合口')) return is('入聲')?'yəʔ':'iʊŋ';
  if (is('庚韻 三等')) return 'iŋ';
  //if (is('清青韻 合口 入聲')) return 'yəʔ'
  if (is('清青韻 合口')) return is('入聲')?'yəʔ':'iʊŋ';
  if (is('清青韻')) return 'iŋ';

  // 曾攝
  if (is('蒸韻 莊組')) return 'əŋ';
  if (is('蒸韻 合口')) return is('入聲')?'yəʔ':'iʊŋ';
  if (is('蒸韻')) return 'iŋ';
  if (is('登韻 合口')) return is('入聲')?'uəʔ':'ʊŋ';
  if (is('登韻 明母 入聲')) return 'əʔ';
  if (is('登韻 幫組')) return is('入聲')?'ʊʔ':'aŋ';
  if (is('登韻')) return 'əŋ';

  // 流攝
  if (is('侯尤韻 幫滂並母 去聲')) return 'u';
  if (is('侯尤韻 並母 上聲')) return 'u';
  if (is('侯韻')) return 'eu';
  if (is('尤韻 幫莊組')) return 'eu';
  if (is('幽韻 幫組')) return 'ieu'
  if (is('尤幽韻')) return 'iu';

  // 深攝
  if (is('侵韻 莊組')) return 'əŋ';
  if (is('侵韻')) return 'iŋ'; 

  // 咸攝
  if (is('談韻 幫組')) return 'ɛn';
  if (is('談韻 舌齒音')) return 'ɛn';
  if (is('覃談韻'))return is('入聲')?'əʔ':'œn';
  if (is('嚴韻 莊組')) return 'ɛn';
  if (is('鹽添嚴韻')) return 'ien';
  if (is('咸銜凡韻')) return 'ɛn';
  
  throw new Error('無韻母規則');
}

function 聲調規則() {{
 
  if (is('全清 或 次清')) {
    if (is('平聲')) return '⁵³'; // 陰平
    if (is('上聲')) return '⁴⁵'; // 陰上
    if (is('去聲')) return '⁴²³'; // 陰去
    if (is('入聲')) return '⁵'; // 陰入
  } else {
    if (is('平聲')) return '²⁴'; // 陽平
    if (is('全濁 上聲')) return '²¹³'; 
    if (is('上聲')) return '³⁵'; // 陽上
    if (is('去聲')) return '²¹³'; // 陽去
    if (is('入聲')) return '²³'; // 陽入
  }
  }
  throw new Error('無聲調規則');
}

let 聲母 = 聲母規則();
let 韻母 = 韻母規則();
let 聲調 = 聲調規則();

if(韻母[0]==="u"&&(聲母==="ɦ"))
 {if(韻母!=="u")韻母 = 韻母.replace('u','');
  	if(聲母==="ɦ")聲母="w";}
if(聲母==="ɦ")
{if(韻母[0]==="i"){聲母="j";if(韻母!=="i"&&韻母!=="iŋ")韻母 = 韻母.replace('i','');}
 if(韻母[0]==="y"){聲母="ɥ";if(韻母!=="y"&&韻母!=="yŋ")韻母 = 韻母.replace('y','');}}

if(!is('入聲'))韻母=韻母.replace('n','');
if(is('入聲'))  
{
     if (韻母.endsWith('n')) {韻母 = 韻母.slice(0, -1) + 'ʔ';}
 else if (韻母.endsWith('ŋ')) {韻母 = 韻母.slice(0, -1) + 'ʔ';韻母= 韻母.replace('iʔ','iəʔ');韻母 = 韻母.replace('yʔ','yəʔ');}
  
}

return 聲母 + 韻母 + 聲調;
