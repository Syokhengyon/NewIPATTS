/* 推導盛唐（平水韻）擬音 —— 精簡設定版
 *
 * 盛唐通用語沒有直接描述音系的材料，但有詩韻（「平水韻」）和日語漢音作爲緊密相關的材料
 *
 * 來源：「平水韻」擬音
 * https://phesoca.com/aws/351/ 或
 * https://www.bilibili.com/read/cv37390491/ 或
 * https://zhuanlan.zhihu.com/p/681190661
 *
 * ── 本版改動 ──
 * 設定項收窄到只剩「合併 / 個別字變化」六項，其餘一律寫死：
 *
 *   非組        → f（f / fʰ / v / ɱ）
 *   等類記法    → 三A Cj ｜ 三BC Cɣ ｜ 一二四 C
 *   見組非三等  → 寫軟腭音（k kʰ ŋ x ɣ，不用小舌 q qʰ ɴ χ ʁ）
 *   支脂元音    → 支 = 脂 i
 *   東韻・流曾攝 元音 → ɨ
 *   魚韻元音    → ɨ
 *   低元音      → 前 a ｜ 後 ɑ
 *   聲調        → 五度數字上標（陰 平⁴² 上⁵⁵ 去³⁴ 入⁴／陽 平²¹ 上²⁴ 去²⁴ 入²）
 *   全濁上      → 歸去，與濁去同值 ²⁴
 *                 （原方案的濁去是 ²²⁴，但本引擎讀不出三值「平-平-升」，實測
 *                  只守住前面平的部分、最後的升到不了；₂²⁴ 已不用，見 README 聲調節）
 *
 * 保留可調的六項：咍泰合併、覃談合併、東一冬合併、輕脣東鍾合併、
 *                 部分蟹攝二等入假攝、部分流攝脣音入遇攝
 *
 * @author unt（原方案）／本專案精簡為固定設定
 */

/** @type { 音韻地位['屬於'] } */
const is = (...x) => 音韻地位.屬於(...x);
/** @type { 音韻地位['判斷'] } */
const when = (...x) => 音韻地位.判斷(...x);

/* ── 寫死的取值（原方案的設定項） ── */
const 非組聲母 = { 幫: 'f', 滂: 'fʰ', 並: 'v', 明: 'ɱ' };
const 等類記法 = { A: 'j', BC: 'ɣ', 其他: '' };   // 三A Cj ｜ 三BC Cɣ ｜ 一二四 C
const 支韻元音 = 'i', 脂韻元音 = 'i';              // 支 = 脂
const 東流曾元音 = 'ɨ';
const 魚韻元音寫法 = 'ɨ';
const 低元音前 = 'a', 低元音後 = 'ɑ';
const 數字用上標 = true;                           // 五度數字用上標
const 全濁上歸去 = true;                           // 全濁上併入去聲（濁上＝濁去 ²⁴）

if (!音韻地位) return [
  '合併',
  ['咍泰合併', false],
  ['覃談合併', true],
  ['東一冬合併', false],
  ['輕脣東鍾合併', false],

  '個別字變化',
  ['部分蟹攝二等入假攝', false],
  ['部分流攝脣音入遇攝', false],
];

function 調整音韻地位() {
  function 調整(表達式, 調整屬性, 字頭串 = null) {
    if (typeof (字頭串) === 'string' && !字頭串.includes(字頭)) return;
    if (is(表達式)) 音韻地位 = 音韻地位.調整(調整屬性);
  }

  // 輕唇化例外
  調整('明母 尤韻', { 等: '一', 類: null, 韻: '侯' });
  調整('明母 東韻', { 等: '一', 類: null });

  // [慧琳反切體現的, 唐代用韻體現的, 據今音推測的]
  const 蟹攝二等入假攝字 = ['崖咼(呙)扠涯搋派差絓畫(画)罣罷(罢)', '佳鼃娃解釵(钗)卦柴', '哇洼蛙灑蝸話(话)掛挂查叉杈衩'].join('');
  const 流攝脣音入遇攝字 = ['浮戊母罦罘蜉矛茂覆懋拇某負(负)阜', '謀(谋)部畝(亩)畮婦(妇)不否桴富牟缶', '復複(复)副牡'].join('');
  if (選項.部分蟹攝二等入假攝 !== false) {
    調整('佳韻 合口 非 上聲', { 韻: '麻' });
    調整('蟹攝 二等', { 韻: '麻' }, 蟹攝二等入假攝字);
  }
  if (選項.部分流攝脣音入遇攝 !== false) {
    調整('幫組 尤侯韻', { 韻: is`尤韻` ? '虞' : '模' }, 流攝脣音入遇攝字);
  }
}

調整音韻地位();

function get聲母() {
  let 聲母 = when([
    ['幫組 C類', 非組聲母[音韻地位.母]],
    ['', {
      幫: 'p', 滂: 'pʰ', 並: 'b', 明: 'm',
      端: 't', 透: 'tʰ', 定: 'd', 泥: 'n', 來: 'l',
      知: 'ʈ', 徹: 'ʈʰ', 澄: 'ɖ', 孃: 'ɳ',
      精: 'ts', 清: 'tsʰ', 從: 'dz', 心: 's', 邪: 'z',
      莊: 'tʂ', 初: 'tʂʰ', 崇: 'dʐ', 生: 'ʂ', 俟: 'ʐ',
      章: 'tɕ', 昌: 'tɕʰ', 常: 'dʑ', 書: 'ɕ', 船: 'ʑ', 日: 'ɲ', 以: 'j',
      見: 'k', 溪: 'kʰ', 羣: 'ɡ', 疑: 'ŋ',
      影: 'ʔ', 曉: 'x', 匣: 'ɣ', 云: '',
    }[音韻地位.母]],
  ]) + when([
    ['幫組 C類', ''],
    ['莊章組 或 日以母', ''],
    ['三等 (A類 或 精組)', 等類記法.A],
    ['三等', 等類記法.BC],
    ['', 等類記法.其他],
  ]);
  return 聲母;
}

function get介音() {
  return is`合口` ? 'w' : '';
}

function get韻基() {
  let 韻基 = when([
    [選項.咍泰合併 === true && '灰咍韻', 'ɑj'], // 不含廢韻
    ['脂之韻', 脂韻元音], ['微韻', [['開口', 'ɨj'], ['', 'uj']]],
    ['齊祭韻', 'ej'], ['灰咍廢韻', 'əj'],
    ['佳皆夬韻', 'aj'], ['泰韻', 'ɑj'],

    ['支韻', 支韻元音], ['魚韻', 魚韻元音寫法], ['虞模韻', 'u'],
    ['麻韻', 'a'], ['歌韻', 'ɑ'],

    ['尤侯幽韻', 東流曾元音 + 'w'],
    ['蕭宵韻', 'ew'],
    ['肴韻', 'aw'], ['豪韻', 'ɑw'],

    ['真臻韻', 'in'], ['殷韻', 'ɨn'], ['文韻', 'un'],
    ['先仙韻', 'en'], ['元魂痕韻', 'ən'],
    ['刪山韻', 'an'], ['寒韻', 'ɑn'],

    ['侵韻', 'im'],
    ['鹽添韻', 'em'], ['嚴凡韻', 'əm'], [選項.覃談合併 === false && '覃韻', 'əm'],
    ['咸銜韻', 'am'], ['覃談韻', 'ɑm'],

    [選項.東一冬合併 === true && '冬韻', 東流曾元音 + 'wŋ'],
    [選項.輕脣東鍾合併 === true && '通攝 幫組 C類', 'uŋ'],
    ['蒸登韻', 東流曾元音 + 'ŋ'], ['東韻', 東流曾元音 + 'wŋ'], ['冬鍾韻', 'uŋ'],
    ['青韻', 'eŋ'],
    ['庚耕清韻', 'aŋ'], ['陽唐韻', 'ɑŋ'], ['江韻', 'awŋ'],
  ]);
  韻基 = 韻基.replace('a', 低元音前);
  韻基 = 韻基.replace('ɑ', 低元音後);
  if (is`入聲`) [...'mnŋ'].forEach((v, i) => { 韻基 = 韻基.replace(v, 'ptk'[i]); });
  return 韻基;
}

function get聲調() {
  const is陰 = is`全清 或 次清 或 次濁 上入聲`;
  let 聲調 = (is陰 ?
    { 平: '⁴²', 上: '⁵⁵', 去: '³⁴', 入: '⁴' } :
    { 平: '²¹', 上: '²⁴', 去: '²⁴', 入: '²' })[全濁上歸去 && is`全濁 上聲` ? '去' : 音韻地位.聲];
  if (數字用上標 === false) 聲調 = [...聲調].map(c => '⁰¹²³⁴⁵'.indexOf(c)).join('');
  return 聲調;
}

function get音節() {
  const 音節 = {
    聲母: get聲母(),
    介音: get介音(),
    韻基: get韻基(),
    聲調: get聲調(),
  };
  if (音節.韻基[0] === 'i') 音節.聲母 = 音節.聲母.replace(/(?<!^)j/, '');
  if (音節.韻基[0] === 'u') 音節.介音 = 音節.介音.replace('w', '');
  音節.韻母 = 音節.介音 + 音節.韻基;
  音節.帶調韻基 = 音節.韻基 + 音節.聲調;
  return 音節;
}

const 音節 = get音節();
return 音節.聲母 + 音節.介音 + 音節.帶調韻基;
